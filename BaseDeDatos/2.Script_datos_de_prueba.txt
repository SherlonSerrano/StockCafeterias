-- ---------------------------------------------------------
-- 9. STOCKLOCAL (75 registros = 3 locales x 25 insumos)
--    Se genera el stock de cada insumo en cada local con una regla
--    sencilla por tipo de insumo. Luego se bajan a proposito
--    8 registros para que existan alertas de stock bajo / sin stock.
-- ---------------------------------------------------------
INSERT INTO StockLocal (IdLocal, IdInsumo, CantidadActual, Ubicacion, FechaUltimoInventario)
SELECT
    l.IdLocal,
    i.IdInsumo,
    CASE
        WHEN i.IdInsumo >= 23 THEN 200 + 50 * l.IdLocal          -- vasos, tapas, sobres
        WHEN i.IdUnidadMedida = 1 THEN 8 + 2 * l.IdLocal + (i.IdInsumo % 4)  -- productos por unidad
        WHEN i.IdUnidadMedida = 2 THEN 6 + 2 * l.IdLocal          -- insumos en kg
        ELSE 20 + 5 * l.IdLocal                                    -- insumos en litros
    END,
    CASE
        WHEN i.IdInsumo <= 17 THEN 'Vitrina'
        WHEN i.IdInsumo <= 22 THEN 'Almacen'
        ELSE 'Mostrador'
    END,
    DATEADD(DAY, -l.IdLocal, SYSDATETIME())
FROM Local l
CROSS JOIN Insumo i;
GO

-- El stock general de cada insumo = suma de lo que hay en los 3 locales
UPDATE Insumo
SET StockActual = (SELECT ISNULL(SUM(sl.CantidadActual), 0)
                   FROM StockLocal sl
                   WHERE sl.IdInsumo = Insumo.IdInsumo);
GO

-- ---------------------------------------------------------
-- 10. MOVIMIENTOSTOCK (30 registros)
--     Se generan 30 filas numeradas (N.i) y para cada una se elige:
--     local, insumo, usuario del mismo local, tipo (E, A, M) y cantidad
--     segun la unidad del insumo. Fechas: los ultimos 30 dias.
-- ---------------------------------------------------------
;WITH N AS (
    SELECT TOP (30) ROW_NUMBER() OVER (ORDER BY (SELECT NULL)) AS i
    FROM sys.all_objects
)
INSERT INTO MovimientoStock (IdLocal, IdInsumo, IdUsuario, TipoMovimiento, Cantidad, FechaMovimiento, Observacion)
SELECT
    ((N.i - 1) % 3) + 1,
    ins.IdInsumo,
    2 + ((N.i - 1) % 3) * 6 + (((N.i - 1) / 3) % 6),
    CASE ((N.i - 1) / 3) % 3 WHEN 0 THEN 'E' WHEN 1 THEN 'A' ELSE 'M' END,
    CASE ins.IdUnidadMedida
        WHEN 1 THEN 1 + (N.i % 10)
        WHEN 2 THEN 2 + (N.i % 5)
        ELSE 5 + (N.i % 5)
    END,
    DATEADD(DAY, -N.i, SYSDATETIME()),
    CONCAT('Movimiento de prueba #', N.i)
FROM N
JOIN Insumo ins ON ins.IdInsumo = ((N.i - 1) % 25) + 1;
GO

-- ---------------------------------------------------------
-- 11. CIERREDIA (21 registros: 7 por local, uno por dia)
-- ---------------------------------------------------------
;WITH N AS (
    SELECT TOP (21) ROW_NUMBER() OVER (ORDER BY (SELECT NULL)) AS i
    FROM sys.all_objects
)
INSERT INTO CierreDia (IdLocal, IdUsuario, FechaCierre, Observacion)
SELECT
    ((N.i - 1) % 3) + 1,
    2 + ((N.i - 1) % 3) * 6 + (((N.i - 1) / 3) % 6),
    DATEADD(DAY, -(((N.i - 1) / 3) + 1), SYSDATETIME()),
    CONCAT('Cierre de prueba #', N.i)
FROM N;
GO

-- ---------------------------------------------------------
-- 12. DETALLECIERRE (63 registros: 3 insumos por cada cierre)
--     En la tercera fila de cada cierre la cantidad esperada es
--     mayor que la contada, para simular una diferencia.
-- ---------------------------------------------------------
INSERT INTO DetalleCierre (IdCierreDia, IdInsumo, CantidadFinal, CantidadEsperada)
SELECT
    cd.IdCierreDia,
    ins.IdInsumo,
    f.Cant,
    f.Cant + CASE WHEN n.Valor = 2 THEN (CASE ins.IdUnidadMedida WHEN 1 THEN 1 ELSE 0.5 END) ELSE 0 END
FROM CierreDia cd
CROSS JOIN (VALUES (0), (1), (2)) n(Valor)
JOIN Insumo ins ON ins.IdInsumo = ((cd.IdCierreDia * 3 + n.Valor) % 25) + 1
CROSS APPLY (
    SELECT CASE ins.IdUnidadMedida
        WHEN 1 THEN 5 + (cd.IdCierreDia % 8)
        WHEN 2 THEN 4 + (cd.IdCierreDia % 5)
        ELSE 10 + (cd.IdCierreDia % 6)
    END AS Cant
) f;
GO

-- ---------------------------------------------------------
-- 13. ALERTA (23 registros, relacionadas a StockLocal)
--     Historial de alertas: se toman los registros de StockLocal
--     cuyo id es multiplo de 3.
--     B = Stock bajo, S = Sin stock
-- ---------------------------------------------------------
;WITH S AS (
    SELECT IdStockLocal, ROW_NUMBER() OVER (ORDER BY IdStockLocal) AS rn
    FROM StockLocal
    WHERE IdStockLocal % 3 = 0
)
INSERT INTO Alerta (IdStockLocal, TipoAlerta, FechaAlerta, Resuelta)
SELECT
    IdStockLocal,
    CASE WHEN rn % 4 = 0 THEN 'S' ELSE 'B' END,
    DATEADD(DAY, -rn, SYSDATETIME()),
    CASE WHEN rn % 3 = 0 THEN 1 ELSE 0 END
FROM S
WHERE rn <= 23;
GO

-- ---------------------------------------------------------
-- VERIFICACION FINAL: conteo de filas por tabla
-- Esperado: Local 3, UnidadMedida 5, CategoriaProducto 5, Rol 3,
-- Producto 23, Insumo 25, Receta 38, Usuario 20, StockLocal 75,
-- MovimientoStock 30, CierreDia 21, DetalleCierre 63, Alerta 23
-- ---------------------------------------------------------
SELECT 'Local' AS Tabla, COUNT(*) AS Filas FROM Local
UNION ALL SELECT 'UnidadMedida', COUNT(*) FROM UnidadMedida
UNION ALL SELECT 'CategoriaProducto', COUNT(*) FROM CategoriaProducto
UNION ALL SELECT 'Rol', COUNT(*) FROM Rol
UNION ALL SELECT 'Producto', COUNT(*) FROM Producto
UNION ALL SELECT 'Insumo', COUNT(*) FROM Insumo
UNION ALL SELECT 'Receta', COUNT(*) FROM Receta
UNION ALL SELECT 'Usuario', COUNT(*) FROM Usuario
UNION ALL SELECT 'StockLocal', COUNT(*) FROM StockLocal
UNION ALL SELECT 'MovimientoStock', COUNT(*) FROM MovimientoStock
UNION ALL SELECT 'CierreDia', COUNT(*) FROM CierreDia
UNION ALL SELECT 'DetalleCierre', COUNT(*) FROM DetalleCierre
UNION ALL SELECT 'Alerta', COUNT(*) FROM Alerta;
GO
